  import { verifySignature } from './verifySignature';
  import { enqueuePaymentEvent } from '../queue/payments';
  export async function processWebhook(req, res) {
    verifySignature(req.headers['x-signature'], req.rawBody);
    await enqueuePaymentEvent(JSON.parse(req.body));
    res.status(202).send('accepted');
  }
